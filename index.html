📋 Project Overview

This JIG Counter is a smart production item counting system built with ESP32-S3. It detects items using an IR sensor, shows real-time count on an OLED display, provides audio and visual feedback, and can control machines via relay when the target count is reached.


Feature Function Status
🔴 IR Sensor Detects each item passing through the jig ✅ Active
🔘 Button 1 Reset counter to zero ✅ Active
🔘 Button 2 Cycle through target count presets ✅ Active
🔊 Buzzer Beep on count + alarm at target ✅ Active
🟢 Green LED Solid = system counting ✅ Active
🔴 Red LED Blinks = target reached ✅ Active
⚡ Relay Stops conveyor/machine at target ✅ Active
📟 OLED 0.96" Displays count, target, progress bar ✅ Active

/*
 * ==========================================
 *  JIG COUNTER - KAIZEN PROJECT
 *  ESP32-S3 Based Production Counter
 * ==========================================
 *  Version: 1.0
 * ==========================================
 */

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ============ PIN DEFINITIONS ============
#define IR_SENSOR_PIN     4
#define BUTTON_RESET_PIN  5
#define BUTTON_SET_PIN    6
#define BUZZER_PIN        7
#define LED_GREEN_PIN     15
#define LED_RED_PIN       16
#define RELAY_PIN         17
#define OLED_SDA          8
#define OLED_SCL          9

// ============ OLED SETTINGS ============
#define SCREEN_WIDTH      128
#define SCREEN_HEIGHT     64
#define OLED_RESET        -1
#define OLED_ADDRESS      0x3C

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// ============ COUNTER VARIABLES ============
volatile unsigned long itemCount = 0;
unsigned long targetCount = 100;
bool targetReached = false;
bool lastIRState = HIGH;

// ============ DEBOUNCE VARIABLES ============
unsigned long lastDebounceIR = 0;
unsigned long lastDebounceBtn1 = 0;
unsigned long lastDebounceBtn2 = 0;
unsigned long debounceDelay = 50;
unsigned long irDebounceDelay = 100;

// ============ TIMING ============
unsigned long lastDisplayUpdate = 0;
unsigned long displayUpdateInterval = 200;

// ============ TARGET PRESETS ============
const unsigned long targetPresets[] = {10, 25, 50, 100, 200, 500, 1000};
const int numPresets = 7;
int currentPresetIndex = 3;

// ============ SETUP ============
void setup() {
  Serial.begin(115200);
  Serial.println("=== JIG COUNTER - KAIZEN PROJECT ===");

  pinMode(IR_SENSOR_PIN, INPUT_PULLUP);
  pinMode(BUTTON_RESET_PIN, INPUT_PULLUP);
  pinMode(BUTTON_SET_PIN, INPUT_PULLUP);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_GREEN_PIN, OUTPUT);
  pinMode(LED_RED_PIN, OUTPUT);
  pinMode(RELAY_PIN, OUTPUT);

  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(LED_GREEN_PIN, HIGH);
  digitalWrite(LED_RED_PIN, LOW);
  digitalWrite(RELAY_PIN, HIGH);

  Wire.begin(OLED_SDA, OLED_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDRESS)) {
    Serial.println("OLED Failed!");
    for (;;) {
      digitalWrite(LED_RED_PIN, !digitalRead(LED_RED_PIN));
      delay(500);
    }
  }

  showSplashScreen();
  delay(2000);
  startupBeep();
  updateDisplay();
  Serial.println("System Ready!");
}

// ============ MAIN LOOP ============
void loop() {
  handleIRSensor();
  handleResetButton();
  handleSetButton();
  handleLEDs();

  if (millis() - lastDisplayUpdate > displayUpdateInterval) {
    updateDisplay();
    lastDisplayUpdate = millis();
  }
}

// ============ IR SENSOR ============
void handleIRSensor() {
  if (targetReached) return;

  bool currentIRState = digitalRead(IR_SENSOR_PIN);

  if (currentIRState == LOW && lastIRState == HIGH) {
    if (millis() - lastDebounceIR > irDebounceDelay) {
      lastDebounceIR = millis();
      itemCount++;

      Serial.print("Count: ");
      Serial.print(itemCount);
      Serial.print(" / ");
      Serial.println(targetCount);

      countBeep();

      if (itemCount >= targetCount) {
        targetReached = true;
        targetReachedAction();
      }
      updateDisplay();
    }
  }
  lastIRState = currentIRState;
}

// ============ RESET BUTTON ============
void handleResetButton() {
  static bool lastBtnState = HIGH;
  bool currentBtnState = digitalRead(BUTTON_RESET_PIN);

  if (currentBtnState == LOW && lastBtnState == HIGH) {
    if (millis() - lastDebounceBtn1 > debounceDelay) {
      lastDebounceBtn1 = millis();
      itemCount = 0;
      targetReached = false;
      digitalWrite(RELAY_PIN, HIGH);
      digitalWrite(LED_GREEN_PIN, HIGH);
      digitalWrite(LED_RED_PIN, LOW);
      digitalWrite(BUZZER_PIN, LOW);
      resetBeep();
      Serial.println(">>> COUNTER RESET <<<");
      updateDisplay();
    }
  }
  lastBtnState = currentBtnState;
}

// ============ SET BUTTON ============
void handleSetButton() {
  static bool lastBtnState = HIGH;
  bool currentBtnState = digitalRead(BUTTON_SET_PIN);

  if (currentBtnState == LOW && lastBtnState == HIGH) {
    if (millis() - lastDebounceBtn2 > debounceDelay) {
      lastDebounceBtn2 = millis();
      currentPresetIndex = (currentPresetIndex + 1) % numPresets;
      targetCount = targetPresets[currentPresetIndex];

      if (itemCount < targetCount) {
        targetReached = false;
        digitalWrite(RELAY_PIN, HIGH);
        digitalWrite(LED_RED_PIN, LOW);
        digitalWrite(LED_GREEN_PIN, HIGH);
      }

      setBeep();
      Serial.print("Target: ");
      Serial.println(targetCount);
      showTargetSetting();
      delay(800);
      updateDisplay();
    }
  }
  lastBtnState = currentBtnState;
}

// ============ TARGET REACHED ============
void targetReachedAction() {
  Serial.println("!!! TARGET REACHED !!!");
  digitalWrite(RELAY_PIN, LOW);
  digitalWrite(LED_RED_PIN, HIGH);
  digitalWrite(LED_GREEN_PIN, LOW);
  alarmBeep();
}

// ============ LED HANDLER ============
void handleLEDs() {
  if (targetReached) {
    static unsigned long lastBlink = 0;
    if (millis() - lastBlink > 300) {
      digitalWrite(LED_RED_PIN, !digitalRead(LED_RED_PIN));
      lastBlink = millis();
    }
    digitalWrite(LED_GREEN_PIN, LOW);
  } else {
    digitalWrite(LED_GREEN_PIN, HIGH);
    digitalWrite(LED_RED_PIN, LOW);
  }
}

// ============ UPDATE DISPLAY ============
void updateDisplay() {
  display.clearDisplay();

  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(10, 0);
  display.print("JIG COUNTER");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  display.setTextSize(1);
  display.setCursor(0, 14);
  display.print("COUNT:");
  display.setTextSize(2);
  display.setCursor(10, 24);

  char countStr[8];
  sprintf(countStr, "%04lu", itemCount);
  display.print(countStr);

  display.setTextSize(1);
  display.setCursor(75, 14);
  display.print("TGT:");
  display.setCursor(75, 24);
  sprintf(countStr, "%04lu", targetCount);
  display.print(countStr);

  display.drawRect(0, 40, 128, 10, SSD1306_WHITE);
  float progress = (float)itemCount / (float)targetCount;
  if (progress > 1.0) progress = 1.0;
  int barWidth = (int)(progress * 124);
  display.fillRect(2, 42, barWidth, 6, SSD1306_WHITE);

  display.drawLine(0, 52, 128, 52, SSD1306_WHITE);
  display.setCursor(0, 55);

  if (targetReached) {
    display.print(">> TARGET REACHED! <<");
  } else {
    display.print("STATUS: COUNTING...");
  }

  display.display();
}

// ============ SPLASH SCREEN ============
void showSplashScreen() {
  display.clearDisplay();
  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(20, 5);
  display.println("KAIZEN");
  display.setTextSize(1);
  display.setCursor(15, 28);
  display.println("JIG COUNTER v1.0");
  display.drawLine(0, 40, 128, 40, SSD1306_WHITE);
  display.setCursor(20, 48);
  display.println("Initializing...");
  display.display();
}

// ============ TARGET SETTING SCREEN ============
void showTargetSetting() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(15, 5);
  display.println("SET TARGET");
  display.drawLine(0, 15, 128, 15, SSD1306_WHITE);
  display.setTextSize(3);
  display.setCursor(15, 25);
  char targetStr[8];
  sprintf(targetStr, "%04lu", targetCount);
  display.print(targetStr);
  display.setTextSize(1);
  display.setCursor(5, 55);
  display.print("Press SET to change");
  display.display();
}

// ============ BUZZER FUNCTIONS ============
void countBeep() {
  digitalWrite(BUZZER_PIN, HIGH);
  delay(30);
  digitalWrite(BUZZER_PIN, LOW);
}

void resetBeep() {
  for (int i = 0; i < 2; i++) {
    digitalWrite(BUZZER_PIN, HIGH); delay(100);
    digitalWrite(BUZZER_PIN, LOW);  delay(50);
  }
}

void setBeep() {
  digitalWrite(BUZZER_PIN, HIGH); delay(50);
  digitalWrite(BUZZER_PIN, LOW);  delay(30);
  digitalWrite(BUZZER_PIN, HIGH); delay(50);
  digitalWrite(BUZZER_PIN, LOW);
}

void startupBeep() {
  for (int i = 0; i < 3; i++) {
    digitalWrite(BUZZER_PIN, HIGH); delay(80);
    digitalWrite(BUZZER_PIN, LOW);  delay(80);
  }
}

void alarmBeep() {
  for (int i = 0; i < 5; i++) {
    digitalWrite(BUZZER_PIN, HIGH); delay(200);
    digitalWrite(BUZZER_PIN, LOW);  delay(100);
  }
}

⚠️ Troubleshooting Guide

❌ OLED not displaying anything
✅ Check I2C address: try 0x3C or 0x3D | Verify SDA=GPIO8, SCL=GPIO9 | Check OLED VCC is 3.3V NOT 5V | Run I2C scanner sketch
❌ IR sensor not detecting items
✅ Adjust sensitivity potentiometer on sensor module | Check detection distance (2-30cm range) | Ensure object is not transparent | Verify OUT wire connected to GPIO4
❌ Double counting (2 counts per item)
✅ Increase irDebounceDelay to 150 or 200ms | Check if sensor is vibrating | Clean sensor lens
❌ Buzzer not working
✅ Check buzzer polarity (+ to GPIO7) | Ensure it's an ACTIVE buzzer not passive | Test with simple blink-style sketch
❌ Relay not clicking
✅ Verify relay is powered by 5V | Check if relay is active LOW (IN goes LOW to activate) | Listen for click sound | Check VCC and GND connections
❌ ESP32 keeps restarting
✅ Use separate 5V supply for relay | Check power supply amperage (minimum 2A) | Check for short circuits on breadboard
❌ Buttons not responding
✅ Verify button wiring (one pin to GPIO, other to GND) | Increase debounceDelay to 100ms | Test with multimeter for continuity
❌ LEDs not lighting up
✅ Check LED polarity (longer leg = anode = +) | Verify 220Ω resistor is present | Check GPIO pin number matches code
❌ Upload fails / Port not found
✅ Hold BOOT button while clicking upload | Install CH340 or CP2102 USB driver | Try different USB cable (data cable, not charge-only)
